cmake_minimum_required(VERSION 3.20)
project(mini_ai_cpp20 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(mini_ai STATIC src/mini_ai.cpp)
target_include_directories(mini_ai PUBLIC include)
target_compile_options(mini_ai PRIVATE -Wall -Wextra -Wpedantic)
add_executable(mini_ai_train src/apps/train.cpp); target_link_libraries(mini_ai_train PRIVATE mini_ai)
add_executable(mini_ai_test src/apps/test.cpp); target_link_libraries(mini_ai_test PRIVATE mini_ai)
enable_testing(); add_executable(mini_ai_unit tests/unit.cpp); target_link_libraries(mini_ai_unit PRIVATE mini_ai); add_test(NAME unit COMMAND mini_ai_unit)
